\section*{Gaussian Mixture Models (GMM)}

\subsection*{The Gaussian}
\begin{keeptogether}
A Gaussian (normal distribution) with mean $\mu$ and variance $\sigma^2$ has the familiar bell shape. In $D$ dimensions, the variance becomes a $D \times D$ covariance matrix $\Sigma$:
\[
\mathcal{N}(x \mid \mu, \sigma^2) = \frac{1}{\sqrt{2\pi\sigma^2}}\, e^{-\frac{(x - \mu)^2}{2\sigma^2}}
\qquad
\mathcal{N}(\mathbf{x} \mid \boldsymbol{\mu}, \Sigma) = \frac{1}{(2\pi)^{D/2}|\Sigma|^{1/2}}\, e^{-\frac{1}{2}(\mathbf{x} - \boldsymbol{\mu})^\top \Sigma^{-1} (\mathbf{x} - \boldsymbol{\mu})}
\]
From the MLE section, fitting one Gaussian gives $\mu_{\text{ML}}$ = the sample mean and $\sigma^2_{\text{ML}}$ = the average squared distance from it. The examples below are 1D.
\end{keeptogether}

\subsection*{Why Mixtures?}
\begin{keeptogether}
Real data often has several bumps (\textbf{modes}), and one Gaussian cannot fit it. A \textbf{Gaussian mixture model} adds up $K$ Gaussians instead, one per bump. Our running example is $N = 8$ points: $1,\ 2,\ 2.5,\ 3,\ 4.5,\ 6.5,\ 7,\ 8$.

\begin{keeptogether}
Each component $k$ has its own mean $\mu_k$, variance $\sigma_k^2$, and \textbf{weight} $\pi_k$ (the fraction of the data it explains):
\[
p(x) = \sum_{k=1}^{K} \pi_k\, \mathcal{N}(x \mid \mu_k, \sigma_k^2)
\qquad \sum_{k=1}^{K} \pi_k = 1
\]
For the fit above: $\pi = (0.50, 0.50)$, $\mu = (2.16, 6.50)$, $\sigma^2 = (0.62, 1.69)$.
\end{keeptogether}

\subsection*{The Hidden Cluster Label}
\begin{keeptogether}
Each point $x_n$ came from one component, but we don't observe which. We call that component the \textbf{latent} (hidden) variable $z_n \in \{1, \dots, K\}$. The GMM is a \textbf{generative model}: it describes how to produce new data in two steps:

{\centering
\begin{tikzpicture}[>=latex, box/.style={draw=gray!60, rounded corners=3pt, align=center, font=\small, inner sep=5pt}]
\node[box] (z) at (0,0) {\textbf{1. Pick a cluster}\\[2pt] $z_n = k$ with probability $\pi_k$};
\node[box] (x) at (6.2,0) {\textbf{2. Sample the point}\\[2pt] $x_n \sim \mathcal{N}(\mu_{z_n}, \sigma_{z_n}^2)$};
\draw[->, thick, gray] (z) -- (x);
\node[font=\scriptsize, green!50!black] at (0,-1.0) {e.g.\ $z_n = 2$ (probability 0.50)};
\node[font=\scriptsize, green!50!black] at (6.2,-1.0) {e.g.\ $x_n = 7.1$ from $\mathcal{N}(6.50, 1.69)$};
\end{tikzpicture}
\par}
\end{keeptogether}

\begin{keeptogether}
Fitting runs this backwards: given $x_n$, how likely is each cluster? Bayes' theorem gives the \textbf{responsibility} $\gamma_{nk}$, the probability that cluster $k$ produced point $n$:
\[
\gamma_{nk} = p(z_n = k \mid x_n) = \frac{\pi_k\, \mathcal{N}(x_n \mid \mu_k, \sigma_k^2)}{\sum_{j=1}^{K} \pi_j\, \mathcal{N}(x_n \mid \mu_j, \sigma_j^2)}
\]
Each point's responsibilities sum to 1, so they act like a \textbf{soft} cluster assignment.
\end{keeptogether}

\subsection*{Why We Need EM}
\begin{keeptogether}
MLE would maximize the log-likelihood, but the log now sits outside a sum:
\[
\ln p(\mathcal{D}) = \sum_{n=1}^{N} \ln \underbrace{\sum_{k=1}^{K} \pi_k\, \mathcal{N}(x_n \mid \mu_k, \sigma_k^2)}_{\makebox[0pt]{\scriptsize log of a sum: does not split apart}}
\]
Setting the derivative to zero gives $\mu_k = \frac{\sum_n \gamma_{nk} x_n}{\sum_n \gamma_{nk}}$, but $\gamma_{nk}$ itself depends on $\mu_k$, so there is no closed form. The problem is circular:
\begin{itemize}
  \item \textbf{Knowing the parameters}, we could compute the responsibilities.
  \item \textbf{Knowing the responsibilities}, we could compute the parameters.
\end{itemize}
The \textbf{expectation-maximization (EM)} algorithm breaks the circle by alternating between the two, the same way K-means alternates between assignments and centers.
\end{keeptogether}

\textbf{Step 1 (E-step): Compute the responsibilities.} Hold the parameters fixed and compute $\gamma_{nk}$ for every point. For $x = 3$:
\[

\]

\begin{keeptogether}
\textbf{Step 2 (M-step): Update the parameters.} Hold the responsibilities fixed and refit each Gaussian, counting each point by its responsibility. These are the single-Gaussian MLE formulas, weighted by $\gamma_{nk}$:
\[
N_k = \sum_{n} \gamma_{nk} \qquad \pi_k = \frac{N_k}{N} \qquad \mu_k = \frac{1}{N_k}\sum_{n} \gamma_{nk}\, x_n \qquad \sigma_k^2 = \frac{1}{N_k}\sum_{n} \gamma_{nk}\,(x_n - \mu_k)^2
\]
$N_k$ is the ``effective'' number of points in cluster $k$. For cluster 1: $N_1 = 3.72$, so $\pi_1 = \tfrac{3.72}{8} = 0.47$, $\mu_1 = 2.07$, and $\sigma_1^2 = 0.54$. Both Gaussians move onto their clusters:

\subsection*{Convergence and Pitfalls}
\begin{keeptogether}
Each EM round \textbf{never lowers the likelihood}, so the algorithm always converges. Like K-means, it has limits:
\begin{itemize}
  \item \textbf{Local optimum:} EM finds a local maximum, so different starting points can give different answers. Starting from K-means centers is standard.
  \item \textbf{Variance collapse:} if one Gaussian shrinks onto a single point, $\sigma_k^2 \to 0$ and the likelihood goes to $\infty$. Implementations add a small minimum variance to prevent this.
\end{itemize}
\end{keeptogether}

\begin{keeptogether}
The two algorithms line up step for step:
\begin{itemize}
  \item \textbf{Assignments:} K-means picks the nearest center; the GMM computes responsibilities $\gamma_{nk}$.
  \item \textbf{Updates:} K-means moves each center to the mean of its points; the GMM refits $\mu_k$, $\sigma_k^2$ and $\pi_k$ with weighted averages.
  \item \textbf{Cluster shape:} K-means clusters have no width or weight; each GMM cluster has its own variance and weight.
\end{itemize}
\end{keeptogether}
