\section*{Gaussian Mixture Models (GMM)}

\subsection*{From K-Means to GMM}
\begin{keeptogether}
K-means gives every point exactly one color and treats each cluster as a round blob around a center. A \textbf{Gaussian mixture model} keeps the same idea but softens it in two ways:
\begin{itemize}
  \item \textbf{Probability assignments:} each point gets a probability for every cluster instead of a single color.
  \item \textbf{Clusters with a shape:} each cluster is a Gaussian with its own mean $\boldsymbol{\mu}_k$, covariance $\Sigma_k$ (its width and tilt), and weight $\pi_k$ (the fraction of the data it explains).
\end{itemize}

\subsection*{The Model}
\begin{keeptogether}
The data's distribution is a weighted sum of $K$ Gaussians, one per cluster (the shaded ellipses above):
\[
p(\mathbf{x}) = \sum_{k=1}^{K} \pi_k\, \mathcal{N}(\mathbf{x} \mid \boldsymbol{\mu}_k, \Sigma_k)
\qquad \sum_{k=1}^{K} \pi_k = 1
\]
As in K-means, each point belongs to one cluster $z_n$, but now $z_n$ is \textbf{hidden}: the model picks $z_n = k$ with probability $\pi_k$, then draws $\mathbf{x}_n$ from that cluster's Gaussian.
\end{keeptogether}

\begin{keeptogether}
\textbf{Step 0: Initialize.} Pick starting parameters, often the K-means centers with round covariances and equal weights $\pi_k = \tfrac{1}{K}$.

\textbf{Step 1 (E-step): Update assignments.} Instead of picking the nearest center, Bayes' theorem gives each point a \textbf{responsibility} $\gamma_{nk}$, the probability that cluster $k$ produced it:
\[
\gamma_{nk} = \frac{\pi_k\, \mathcal{N}(\mathbf{x}_n \mid \boldsymbol{\mu}_k, \Sigma_k)}{\sum_{j=1}^{K} \pi_j\, \mathcal{N}(\mathbf{x}_n \mid \boldsymbol{\mu}_j, \Sigma_j)}
\]
\end{keeptogether}

\begin{keeptogether}
\textbf{Step 2 (M-step): Update clusters.} This is K-means' center update, with each point counted by its responsibility instead of fully or not at all:
\[
\underbrace{\boldsymbol{\mu}_k = \frac{1}{N_k}\sum_{n \,:\, z_n = k} \mathbf{x}_n}_{\makebox[0pt]{\scriptsize K-means}}
\qquad\Longrightarrow\qquad
\underbrace{\boldsymbol{\mu}_k = \frac{1}{N_k}\sum_{n} \gamma_{nk}\, \mathbf{x}_n}_{\makebox[0pt]{\scriptsize GMM}}
\qquad N_k = \sum_n \gamma_{nk}
\]
The GMM also updates each cluster's shape and weight the same way:
\[
\Sigma_k = \frac{1}{N_k}\sum_{n} \gamma_{nk}\,(\mathbf{x}_n - \boldsymbol{\mu}_k)(\mathbf{x}_n - \boldsymbol{\mu}_k)^\top \qquad \pi_k = \frac{N_k}{N}
\]

\textbf{Step 3: Repeat} Steps 1 and 2 until nothing changes. Like K-means, EM always converges, but only to a local optimum, so starting from K-means centers is standard.
\end{keeptogether}

\begin{keeptogether}
In 1D, each step is quick to compute by hand. With two clusters at $\mu = (0, 4)$, $\sigma^2 = 1$ and $\pi = (0.5, 0.5)$, the E-step for the point $x = 1$:
\[
\gamma_{n1} = \frac{0.5\,\mathcal{N}(1 \mid 0, 1)}{0.5\,\mathcal{N}(1 \mid 0, 1) + 0.5\,\mathcal{N}(1 \mid 4, 1)} = \frac{0.242}{0.242 + 0.004} = 0.98
\]
An M-step for points $x = 0, 1, 3, 4$ with $\gamma_{n1} = 0.9, 0.8, 0.2, 0.1$:
\[
N_1 = 2 \qquad \pi_1 = \frac{2}{4} = 0.5 \qquad \mu_1 = \frac{0.9(0) + 0.8(1) + 0.2(3) + 0.1(4)}{2} = 0.9
\]
\end{keeptogether}
