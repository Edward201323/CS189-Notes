\section*{ML Basics \& K-Nearest Neighbors}

\subsection*{Train / Validation / Test Split}
Example: 1000 labeled points $\rightarrow$ 600 train / 200 validation / 200 test.

\begin{itemize}
  \item \textbf{Train}: the algorithm uses this data to find the best weights $\mathbf{w}$ (e.g.\ the linear regression line).
  \item \textbf{Validation}: we use this data to compare models and pick the best hyperparameters.
  \item \textbf{Test}: we use this data once at the end to see how well the final model does on data it has never seen.
\end{itemize}

\subsection*{K-Nearest Neighbors (KNN)}
\textbf{Idea:} to classify a new point, find the $k$ closest training points and take a majority vote.

\end{tikzpicture}
\par}

\begin{itemize}
  \item The circle holds the 3 nearest neighbors: 2 red and 1 blue, so the new point is predicted \textbf{red}.
  \item Change $k$ and the answer can change. A bigger circle may pull in more blue points.
  \item \textbf{Confidence} = the share of neighbors that voted for the winning class (here, 2 out of 3).
  \item \textbf{No training step}: KNN just stores the data. Prediction is the slow part, because it measures the distance to every training point.
  \item \textbf{Standardize features first.} Otherwise the feature with the biggest units (e.g.\ pounds vs.\ inches) dominates the distance.
\end{itemize}

\section*{K-Means Clustering}

\subsection*{Clustering}
\textbf{Goal:} split the data into $K$ groups (clusters) of points that are close together. In the plot above, an algorithm should discover the 2 groups on its own.
\begin{itemize}
  \item \textbf{Most common use: market segmentation.} Given a database of customers, group them by e.g.\ age/gender so each group can get its own marketing campaign.
  \item The first clustering algorithm is \textbf{K-means}. $K$ = number of clusters (a hyperparameter we pick).
\end{itemize}

\subsection*{K-Means (Lloyd's Algorithm)}
\textbf{Notation.}
\begin{itemize}
  \item $\boldsymbol{\mu}_k \in \mathbb{R}^D$: the \textbf{cluster center} (centroid) of cluster $k$, for $k = 1, \dots, K$. These are the crosses.
  \item $z_n \in \{1, \dots, K\}$: the \textbf{assignment} of point $\mathbf{x}_n$, i.e.\ which cluster (color) it belongs to.
\end{itemize}

\textbf{Step 0: Initialize.} Pick $K$ starting centers $\boldsymbol{\mu}_1, \dots, \boldsymbol{\mu}_K$ at random (common choice: $K$ random training points).

\textbf{Then repeat (K-means is iterative):}

\textbf{Step 1: Update assignments.} Go through every training example and color it by whichever center is closest:
\[
z_n = \arg\min_{k} \; \|\mathbf{x}_n - \boldsymbol{\mu}_k\|_2^2
\qquad \text{for every } n = 1, \dots, N
\]

\textbf{Step 2: Update centers.} Move each cross to the mean (centroid) of the points with its color:
\[
\boldsymbol{\mu}_k = \frac{1}{N_k} \sum_{n \,:\, z_n = k} \mathbf{x}_n
\qquad N_k = \text{number of points assigned to cluster } k
\]

\subsection*{Cost Function}
Total squared distance from each point to its own center. K-means finds the $z$ and $\boldsymbol{\mu}$ that minimize it:
\[
J(z, \boldsymbol{\mu}) = \sum_{n=1}^{N} \|\mathbf{x}_n - \boldsymbol{\mu}_{z_n}\|_2^2
\]
\begin{itemize}
  \item Small $J$ = tight clusters. E.g.\ points $1, 2, 9, 10$: clusters $\{1,2\},\{9,10\}$ give $J = 1$; clusters $\{1\},\{2,9,10\}$ give $J = 38$.
  \item Step 1 lowers $J$ by changing the colors $z$; Step 2 lowers $J$ by moving the centers $\boldsymbol{\mu}$.
  \item \textbf{Converges:} each step can only lower $J$ (or keep it the same), and $J \geq 0$, so $J$ settles at a final value.
  \item \textbf{Local minima:} that final value depends on the starting centers. Fix: run K-means several times with different random starts and keep the lowest $J$.
\end{itemize}

\subsection*{Choosing $K$}
The right number of clusters is often ambiguous. Don't just minimize $J$: it always shrinks as $K$ grows ($K = N \Rightarrow J = 0$).
\begin{itemize}
  \item \textbf{Domain understanding:} e.g.\ 3 penguin species $\Rightarrow K = 3$; 4 marketing campaigns $\Rightarrow K = 4$.
  \item \textbf{Elbow method:} plot $J$ vs.\ $K$ and pick where the improvement levels off.
\end{itemize}

\subsection*{The Supervised Learning Pipeline}
\[
\text{Training set } \mathcal{D} \;\longrightarrow\; \text{Learning algorithm} \;\longrightarrow\; \text{Hypothesis } y(\mathbf{x}, \mathbf{w})
\]
\begin{itemize}
  \item The learning algorithm takes the training set and outputs a function (the \textbf{hypothesis}).
  \item The hypothesis takes a house we have not seen yet and outputs an estimated price.
\end{itemize}

\subsection*{Representing the Hypothesis}
One feature ($x_1$ = size):
\[
y(\mathbf{x}, \mathbf{w}) = w_0 + w_1 x_1
\]

Two features: add number of bedrooms.

\begin{center}
\begin{tabular}{ccc}
\toprule
Size $x_1$ & \# Bedrooms $x_2$ & Price $t$ \\
\midrule
2104 & 3 & 400 \\
1416 & 2 & 232 \\
1534 & 3 & 315 \\
852  & 2 & 178 \\
\bottomrule
\end{tabular}
\end{center}

\[
y(\mathbf{x}, \mathbf{w}) = w_0 + w_1 x_1 + w_2 x_2
\]

\textbf{Dummy feature trick:} define $x_0 = 1$ always, so the bias $w_0$ has a feature to multiply and fits inside one dot product $\mathbf{w}^\top\mathbf{x}$. Then
\[
y(\mathbf{x}, \mathbf{w}) = \sum_{d=0}^{D} w_d x_d = \mathbf{w}^\top \mathbf{x},
\qquad
\mathbf{x} = \begin{bmatrix} 1 \\ x_1 \\ x_2 \end{bmatrix},\;
\mathbf{w} = \begin{bmatrix} w_0 \\ w_1 \\ w_2 \end{bmatrix}
\]
Example: $\mathbf{w} = [50,\, 0.1,\, 20]^\top$, house $(2104, 3)$:
$\mathbf{w}^\top\mathbf{x} = 50(1) + 0.1(2104) + 20(3) = 320.4$.

\subsection*{The Loss (Least Squares / OLS)}
\[
E(\mathbf{w}) = \frac{1}{2}\sum_{n=1}^{N} \Big( \underbrace{y(\mathbf{x}_n, \mathbf{w})}_{\text{prediction}} - \underbrace{t_n}_{\text{target}} \Big)^2
\]
\begin{itemize}
  \item Squared difference between prediction and true value, summed over all $N$ examples.
  \item The $\tfrac12$ only cancels the 2 when differentiating; it does not change the minimizer.
\end{itemize}

We want the weights that minimize the loss, $\mathbf{w}^* =  \arg\min_{\mathbf{w}} E(\mathbf{w})$. We can find them using \textbf{gradient descent}.

\section*{Gradient Descent}

\subsection*{Batch Gradient Descent}
\textbf{Batch} = every step uses \emph{all} $N$ training examples to compute the gradient.

Start with some weights $\mathbf{w}$, usually all zeros. Then repeat:
\begin{itemize}
  \item Look around the current point and find the steepest \emph{downhill} direction.
  \item Take a small step in that direction.
  \item Stop when the error stops going down.
\end{itemize}

\begin{itemize}
  \item The step size is called the \textbf{learning rate}: too big overshoots, too small is slow.
  \item Depending on where you start, you can end up in a \textbf{local minimum} instead of the global one.
  \item For linear regression, the error is a quadratic function, so there is only one minimum.
\end{itemize} 

\textbf{Another view:} each step changes $\mathbf{w}$, which changes the line. Starting from $\mathbf{w} = \mathbf{0}$ (a flat line at 0), every step fits the data a little better until it converges.

\textbf{Update rule.} Repeat until convergence, for every $d = 0, 1, \dots, D$:
\[
w_d := w_d - \eta \, \frac{\partial E(\mathbf{w})}{\partial w_d}
\]
\begin{itemize}
  \item $:=$ means assignment (overwrite $w_d$ with the right side).
  \item $\eta$: learning rate. The partial derivative points uphill, so we subtract it.
\end{itemize}

\textbf{Computing the derivative.} Start with one training example $(\mathbf{x}, t)$. Write $y = y(\mathbf{x},\mathbf{w})$ for short.
\begin{align*}
\frac{\partial}{\partial w_d}\, \frac{1}{2}(y - t)^2
&= (y - t) \cdot \frac{\partial}{\partial w_d}(y - t)
  \qquad \text{(chain rule; the 2 cancels the $\tfrac12$)} \\
&= (y - t) \cdot \frac{\partial}{\partial w_d}\big(w_0 x_0 + \dots + w_d x_d + \dots + w_D x_D - t\big) \\
&= (y - t)\, x_d
  \qquad \text{(only $w_d x_d$ depends on $w_d$)}
\end{align*}

For all $N$ examples, sum (the derivative of a sum is the sum of the derivatives):
\[
\frac{\partial E(\mathbf{w})}{\partial w_d} = \sum_{n=1}^{N} \big(y(\mathbf{x}_n,\mathbf{w}) - t_n\big)\, x_{n,d}
\]

\textbf{Final algorithm.} Repeat until convergence, for every $d$:
\[
w_d := w_d - \eta \sum_{n=1}^{N}
\underbrace{\big(y(\mathbf{x}_n,\mathbf{w}) - t_n\big)}_{\text{error on example } n}\,
\underbrace{x_{n,d}}_{\text{feature } d}
\]
\begin{itemize}
  \item The $\sum_{n=1}^{N}$ is what makes it \textbf{batch}: all $N$ examples, every step.
  \item Intuition: each weight moves by (error) $\times$ (its feature). Big error or big feature $\Rightarrow$ big update.
  \item Downside: if $N$ is huge, each step scans the whole dataset, so it's slow.
\end{itemize}


\subsection*{Stochastic Gradient Descent (SGD)}
Instead of scanning through \emph{all} $N$ examples before each update, SGD updates $\mathbf{w}$ using \textbf{one example at a time}. Then repeat:
\begin{itemize}
  \item Pick one training example $(\mathbf{x}_n, t_n)$.
  \item Take a small step that fits \emph{that one example} a little better.
  \item Move on to the next example.
\end{itemize}

\begin{itemize}
  \item The path is \textbf{noisy}: each step fits one example, so it doesn't always head downhill, but it does on average.
  \item It \textbf{never fully converges}. It keeps bouncing around in a region near the minimum.
  \item Fix: \textbf{slowly decrease the learning rate} over time. Smaller steps help it settle closer to the minimum.
  \item Why use it: each step is \emph{much} cheaper, so on large datasets it makes progress far faster than batch GD.
  \item Small dataset: just use batch GD (one less thing to tune). Large dataset: use SGD.
\end{itemize}

\textbf{Update rule.} Loop over $n = 1, \dots, N$; for each $n$, update every $d$:
\[
w_d := w_d - \eta \,
\underbrace{\big(y(\mathbf{x}_n,\mathbf{w}) - t_n\big)}_{\text{error on example } n}\,
\underbrace{x_{n,d}}_{\text{feature } d}
\]
\begin{itemize}
  \item Same as batch GD, but with the $\sum_{n=1}^{N}$ \textbf{dropped}: one example per step.
  \item \textbf{Mini-batch}: use a small batch of $B$ examples per step (e.g.\ 100). $B = 1$ is SGD, $B = N$ is batch GD.
\end{itemize}

\section*{Normal Equation (Closed Form)}
For linear regression we can skip gradient descent and \textbf{jump straight to the minimum}: the bottom of the bowl has zero slope, so set the gradient to $\mathbf{0}$ and solve.

\textbf{Deriving the normal equation:}

\textbf{Step 1: Rewrite the loss into matrix form.}
\[
\mathbb{X} = \begin{bmatrix} \mathbf{x}_1^\top \\ \vdots \\ \mathbf{x}_N^\top \end{bmatrix} \in \mathbb{R}^{N\times(D+1)},
\qquad
\mathbf{t} = \begin{bmatrix} t_1 \\ \vdots \\ t_N \end{bmatrix} \in \mathbb{R}^{N}
\]
\begin{itemize}
  \item $\mathbb{X}$: the \textbf{design matrix}, one row per example (including the dummy 1).
  \item $\mathbf{t}$: all the targets.
  \item $\mathbb{X}\mathbf{w}$: all $N$ predictions at once. $\;\mathbb{X}\mathbf{w} - \mathbf{t}$: all $N$ errors (prediction minus target).
\end{itemize}

The loss is half the sum of squared errors, and a vector times itself is the sum of its squares so:
\[
E(\mathbf{w}) = \frac12\sum_{n=1}^{N}\big(y(\mathbf{x}_n,\mathbf{w}) - t_n\big)^2 = \frac12(\mathbb{X}\mathbf{w} - \mathbf{t})^\top(\mathbb{X}\mathbf{w} - \mathbf{t})
\]

\textbf{Step 2: Set the gradient equal to zero and solve w}:
\begin{align*}
E(\mathbf{w})
&= \frac12\big(\mathbf{w}^\top\mathbb{X}^\top\mathbb{X}\mathbf{w} - 2\,\mathbf{w}^\top\mathbb{X}^\top\mathbf{t} + \mathbf{t}^\top\mathbf{t}\big)
  \qquad \text{(expand like $(a-b)^2 = a^2 - 2ab + b^2$)} \\
\nabla_{\mathbf{w}} E
&= \mathbb{X}^\top\mathbb{X}\mathbf{w} - \mathbb{X}^\top\mathbf{t}
  \qquad \text{(using $\nabla_{\mathbf{w}}\, \mathbf{w}^\top A\mathbf{w} = 2A\mathbf{w}$ and $\nabla_{\mathbf{w}}\, \mathbf{w}^\top\mathbf{u} = \mathbf{u}$)} \\
\mathbf{0}
&= \mathbb{X}^\top\mathbb{X}\mathbf{w} - \mathbb{X}^\top\mathbf{t}
  \qquad \text{(set the gradient to zero)}
\end{align*}
\begin{itemize}
  \item Expanding: the two middle terms $\mathbf{w}^\top\mathbb{X}^\top\mathbf{t}$ and $\mathbf{t}^\top\mathbb{X}\mathbf{w}$ are the same single number, so they combine into $-2\,\mathbf{w}^\top\mathbb{X}^\top\mathbf{t}$.
  \item Gradient: $A = \mathbb{X}^\top\mathbb{X}$ and $\mathbf{u} = \mathbb{X}^\top\mathbf{t}$; $\mathbf{t}^\top\mathbf{t}$ has no $\mathbf{w}$, so it drops. These are the matrix versions of $\tfrac{d}{dw}aw^2 = 2aw$ and $\tfrac{d}{dw}uw = u$.
\end{itemize}
Rearranging gives the \textbf{normal equation} $\mathbb{X}^\top\mathbb{X}\,\mathbf{w} = \mathbb{X}^\top\mathbf{t}$. Multiply both sides by $(\mathbb{X}^\top\mathbb{X})^{-1}$ (the matrix version of dividing):
\[
\mathbf{w}^* = (\mathbb{X}^\top\mathbb{X})^{-1}\mathbb{X}^\top\mathbf{t}
\]

\begin{itemize}
  \item \textbf{Requires $\mathbb{X}^\top\mathbb{X}$ to be invertible.} Fails if $N < D+1$ (fewer examples than weights) or if features are redundant. Fix: ridge regression, $\mathbf{w}^* = (\mathbb{X}^\top\mathbb{X} + \lambda I)^{-1}\mathbb{X}^\top\mathbf{t}$.
  \item \textbf{Normal equation vs.\ gradient descent.} The normal equation is exact and takes one step, but the inverse is expensive for large problems and it only works for linear regression.
  \item \textbf{Geometric view.} $\mathbb{X}\mathbf{w}^*$ is the closest point to $\mathbf{t}$ reachable by the columns of $\mathbb{X}$, so the error is perpendicular to every column: $\mathbb{X}^\top(\mathbb{X}\mathbf{w}^* - \mathbf{t}) = \mathbf{0}$.
\end{itemize}

\section*{Principal Component Analysis (PCA)}

\subsection*{What PCA Does}
\textbf{Goal:} reduce unlabeled data from $D$ to $k \ll D$ dimensions (e.g.\ 10{,}000 $\rightarrow$ 100) while losing as little information as possible.
\begin{itemize}
  \item Find the direction the data is most spread out along, then describe each point by where it lands on that direction.
  \item \textbf{Uses:} visualization (2D/3D plots), compression, and preprocessing for other models.
\end{itemize}

\subsection*{PCA Step by Step}
\textbf{Running example} (2D $\rightarrow$ 1D): 6 points $(0,0),(0,1),(1,0),(1,2),(2,1),(2,2)$.

\textbf{Step 1: Center.} Subtract the mean from every point (the best line passes through the mean):
\[
\bar{\mathbf{x}} = \frac1N\sum_{n=1}^N \mathbf{x}_n = (1,1),
\qquad
\mathbf{x}_n \leftarrow \mathbf{x}_n - \bar{\mathbf{x}}
\]

\textbf{Step 2: Find the eigenvectors.} Stack the centered points as rows of $X$. The eigenvectors of $X^\top X$ are the cloud's long and short axes:
\[
X^\top X = \begin{bmatrix}4&2\\2&4\end{bmatrix},
\qquad
\mathbf{u}_1 = \tfrac{1}{\sqrt2}(1,1),\; \lambda_1 = 6,
\qquad
\mathbf{u}_2 = \tfrac{1}{\sqrt2}(1,-1),\; \lambda_2 = 2
\]

\textbf{Step 3: Keep the top $k$.} Each eigenvalue is the spread along its eigenvector, so keep the largest. Here $k = 1$: keep $\mathbf{u}_1$, drop $\mathbf{u}_2$.

\textbf{Step 4: Project.} Each point becomes how far along $\mathbf{u}_1$ it lands:
\[
z_n = \mathbf{x}_n^\top \mathbf{u}_1,
\qquad \text{e.g.\ } (1,1) \rightarrow \tfrac{1+1}{\sqrt2} = \sqrt2
\]

\textbf{Step 5 (if asked): Reconstruct.} Start at the mean and go $z_n$ along $\mathbf{u}_1$. The gap $\mathbf{x}_n - \hat{\mathbf{x}}_n$ is the error:
\[
\hat{\mathbf{x}}_n = \bar{\mathbf{x}} + z_n\,\mathbf{u}_1
\]

\subsection*{Why the Top Eigenvector}
PCA picks the unit direction $\mathbf{u}$ that keeps the most spread:
\[
\max_{\|\mathbf{u}\|=1} \; \sum_{n=1}^N (\mathbf{x}_n^\top\mathbf{u})^2 = \max_{\|\mathbf{u}\|=1}\; \mathbf{u}^\top X^\top X\,\mathbf{u}
\quad\Longrightarrow\quad
X^\top X\,\mathbf{u} = \lambda\mathbf{u}
\]

\begin{itemize}
  \item \textbf{Max spread = min error.} Each point's length$^2$ = kept$^2$ + lost$^2$ (Pythagoras), so the total is fixed: here kept + lost $= 8$ at every angle.
  \item \textbf{Eigenvalue = spread} along its eigenvector, so the best direction is the one with the largest $\lambda$.
  \item $\|\mathbf{u}\| = 1$ makes $\mathbf{x}^\top\mathbf{u}$ the true projected length. The sign doesn't matter: $\pm\mathbf{u}$ is the same line.
  \item \textbf{Not linear regression:} PCA uses \emph{perpendicular} distance and no labels; regression uses \emph{vertical} distance.
\end{itemize}

\subsection*{Finding Eigenvectors by Hand}
\begin{enumerate}[leftmargin=1.5em]
  \item Eigenvalues: $\det(X^\top X - \lambda I) = (4-\lambda)^2 - 4 = (\lambda - 6)(\lambda - 2) = 0 \;\Rightarrow\; \lambda = 6,\, 2$.
  \item Eigenvectors: solve $(X^\top X - \lambda I)\mathbf{v} = \mathbf{0}$: $\;\lambda = 6 \to (1,1)$, $\;\lambda = 2 \to (1,-1)$.
  \item Normalize to length 1: $\mathbf{u}_1 = \frac{1}{\sqrt2}(1,1)$, $\;\mathbf{u}_2 = \frac{1}{\sqrt2}(1,-1)$.
\end{enumerate}

\subsection*{$k$ Components and Choosing $k$}
Keep the top $k$ eigenvectors $\mathbf{u}_1, \dots, \mathbf{u}_k$:
\[
z_{nj} = (\mathbf{x}_n - \bar{\mathbf{x}})^\top \mathbf{u}_j,
\qquad
\hat{\mathbf{x}}_n = \bar{\mathbf{x}} + \sum_{j=1}^k z_{nj}\,\mathbf{u}_j
\]
\begin{itemize}
  \item E.g.\ $\mathbf{x} \in \mathbb{R}^{10{,}000}$, $k = 10$: $\;\mathbf{x} \to (\mathbf{u}_1^\top\mathbf{x}, \dots, \mathbf{u}_{10}^\top\mathbf{x})$.
  \item \textbf{Error} = sum of the \emph{dropped} eigenvalues.
  \item \textbf{Fraction of variance kept} $= \dfrac{\lambda_1 + \dots + \lambda_k}{\lambda_1 + \dots + \lambda_D}$, e.g.\ $\frac{6}{8} = 75\%$. Pick the smallest $k$ keeping $\sim$90--95\%.
\end{itemize}

\subsection*{Good to Know}
\begin{itemize}
  \item \textbf{Standardize} (divide each feature by its std) when features have different units. Otherwise the biggest units dominate.
  \item \textbf{Vector projection} means the point $(\mathbf{x}^\top\mathbf{u})\,\mathbf{u}$ in the original space, not the number $z$. E.g.\ $(1,2) \to (\frac32, \frac32)$.
  \item \textbf{$\Sigma = \frac1N X^\top X$ vs.\ $X^\top X$:} same eigenvectors; the eigenvalues of $\Sigma$ are the variances ($1, \frac13$), $N\times$ smaller. Ranking and fraction kept don't change.
  \item \textbf{Via SVD:} $X = U S V^\top \Rightarrow X^\top X = V S^2 V^\top$, so the principal components are the columns of $V$, with $\lambda_j = s_j^2/N$.
\end{itemize}
